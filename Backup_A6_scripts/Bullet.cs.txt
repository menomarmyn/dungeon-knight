using UnityEngine;

// На префабе пули нужны Rigidbody2D без гравитации и триггер-коллайдер.
public class Bullet : MonoBehaviour
{
    public float speed = 12f;

    void Start()
    {
        GetComponent<Rigidbody2D>().linearVelocity = transform.right * speed;
        Destroy(gameObject, 3f); // Пуля исчезает через 3 секунды.
    }

    void OnTriggerEnter2D(Collider2D other)
    {
        // Пуля проходит сквозь триггеры и героя.
        if (other.isTrigger || other.CompareTag("Player"))
            return;

        // Опасность сама решает, что делать при попадании.
        Hazard hazard = other.GetComponent<Hazard>();
        if (hazard != null)
            hazard.OnShot();

        // Патрульного врага можно уничтожить выстрелом.
        if (other.GetComponent<EnemyPatrol>() != null)
            Destroy(other.gameObject);

        Destroy(gameObject); // Пуля исчезает при столкновении с преградой.
    }
}
